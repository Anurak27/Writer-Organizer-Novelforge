   105→function formatWordCount(words: number): string {

   541→                            type="file"
   542→                            accept="image/*"

   544→                            onChange={(e) => {
   545→                              const f = e.target.files?.[0];
   546→                              if (f) uploadCover(f, 'create');

   548→                          />
   549→                          <div className={`${inputCls} flex items-center justify-center h-10 border border-dashed border-zinc-700 rounded-md hover:border-zinc-500 transition-colors`}>

   553→                              <>
   554→                                <ImagePlus className="w-4 h-4 text-zinc-500 mr-2" />
   555→                                <span className="text-xs text-zinc-500">Choose cover image</span>
   556→                              </>
   557→                            )}
   558→                          </div>
   559→                        </label>
   560→                      </div>
   561→                    </div>
